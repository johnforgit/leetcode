class Solution {
    public void rotate(int[][] nums) {
        for( int i=0; i<nums.length; i++ ){
                for( int j =i+1;j<nums.length;j++){
                    int temp = nums[i][j];
                    nums[i][j] = nums[j][i];
                    nums[j][i]= temp;
                }
        }
        for( int i=0;i<nums.length;i++){
            int index = nums.length-1;
            for( int j=0;j<nums.length/2;j++){
                int temp = nums[i][j];
                nums[i][j] = nums[i][index];
                nums[i][index]= temp ;
                index--;
            }
        }

    }
}
