class Solution:
    def largestMerge(self, word1: str, word2: str) -> str:
        merge = []
        p1, p2 = 0, 0
        n1, n2 = len(word1), len(word2)
        while p1 < n1 and p2 < n2:
            if word1[p1] > word2[p2]:
                merge.append(word1[p1])
                p1 += 1
            elif word1[p1] < word2[p2]:
                merge.append(word2[p2])
                p2 += 1
            else:
                if word1[p1:] > word2[p2:]:
                    merge.append(word1[p1])
                    p1 += 1
                else:
                    merge.append(word2[p2])
                    p2 += 1
        # print(f"merge: {merge}")

        if p1 < n1:
            merge.append(word1[p1:])
        if p2 < n2:
            merge.append(word2[p2:])

        return "".join(merge)
