class Solution:
    def reverseWords(self, s: str) -> str:
        vowels = set("aeiou")

        # Function to count vowels in a word
        def count_vowels(word):
            return sum(c in vowels for c in word)

        words = s.split()
        first_vowel_count = count_vowels(words[0])

        result = [words[0]]  # first word stays the same

        # Process remaining words
        for w in words[1:]:
            if count_vowels(w) == first_vowel_count:
                result.append(w[::-1])  # reverse the word
            else:
                result.append(w)

        return " ".join(result)

class Solution {
    public String reverseWords(String s) {
        String[] words = s.split(" ");
        if(words.length == 1)
            return s;
        int vowelCountFirst = countVowels(words[0]);
        StringBuilder result = new StringBuilder();
        result.append(words[0]);
        result.append(" ");
        for(int i=1; i<words.length; i++) {
            // find the vowelcount of our word
            int wordVowelCount = countVowels(words[i]);

            // check if the word has the same vowel count as that of our
            // first word
            if(vowelCountFirst == wordVowelCount) {
                result.append(reverseWord(words[i]));
            } else {
                result.append(words[i]);
            }

            // add space except after the last word
            if(i < (words.length - 1)) {
                result.append(" ");
            }
        }
        return result.toString();
    }

    public String reverseWord(String word) {
        String reversed = new StringBuilder(word).reverse().toString();
        return reversed;
    }

    public int countVowels(String word) {
        int count = 0;
        word = word.toLowerCase();

        for (int i = 0; i < word.length(); i++) {
            char ch = word.charAt(i);
            if (ch == 'a' || ch == 'e' || ch == 'i' || ch == 'o' 
                || ch == 'u') {
                count++;
            }
        }
        return count;
    }
}


public String reverseWords(String s) {
        Set<Character> vowels = new HashSet<>(Arrays.asList('a','e','i','o','u'));
        int gc = -1;
        char[] arr = s.toCharArray();
        int i = 0;
        while (i < arr.length) {
            int c = 0;
            int start = i;
            while (i < arr.length && arr[i] != ' ') {
                if (vowels.contains(arr[i])) c++;
                i++;
            }
            if (c == gc) {
                reverse(arr, start, i - 1);
            }
            if (gc == -1) gc = c;
            i++;
        }
        return new String(arr);
    }
    
private void reverse(char[] arr, int l, int r) {
	while (l < r) {
		char temp = arr[l];
		arr[l] = arr[r];
		arr[r] = temp;
		l++;
		r--;
    }
}


class Solution {
    int isvowel(char[] ori,int s,int e)
            {
          int c=0;
            for(int i=s;i<=e;i++)
                {
                    if((char)(ori[i])=='a'||(char)(ori[i])=='e'||(char)(ori[i])=='i'||(char)(ori[i])=='o'||(char)(ori[i])=='u')
                    {
                        c++;
                    }
                }
          return c;
            }
	void reverse(char[] ori,int s,int e)
    {
        for(int i=s;i<=(s+e)/2;i++)
            {
                char temp=ori[i];
                ori[i]=ori[s+e-i];
                ori[s+e-i]=temp;
            }
    }
	
    public String reverseWords(String s) 
	{
        char[] ori=s.toCharArray();
        int n=ori.length;
        int res=0,i;
        for(i=0;i<n&&ori[i]!=' ';i++)
            {
                if((char)(ori[i])=='a'||(char)(ori[i])=='e'||(char)(ori[i])=='i'||(char)(ori[i])=='o'||(char)(ori[i])=='u')
                {
                    res++;
                }
            }
        for(int j=i; j<n; ) {
			while(j < n && ori[j] == ' ')
				j++;
			if(j >= n) break;
			int s1 = j;
			while(j < n && ori[j] != ' ')
				j++;
			int val = isvowel(ori, s1, j-1);
			if(val == res)
			{
				reverse(ori, s1, j-1);
			}
		}
		return new String(ori);
    }
}
