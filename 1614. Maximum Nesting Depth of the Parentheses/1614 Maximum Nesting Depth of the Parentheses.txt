    // USING A COUNTER VARIABLE
    public int maxDepth(String s) {
        int ans = 0;
        int openBrackets = 0;

        for(Character c : s.toCharArray()) {
            if(c == '(') {
                openBrackets++;
            } else if(c == ')') {
                openBrackets--;
            }

            ans = Math.max(ans, openBrackets);
        }

        return ans;
    }


     public int maxDepth(String s) {
		int cnt = 0;
		int max = 0;
        for(int i=0;i<s.length();i++){
            char ch = s.charAt(i);
            if(ch=='('){
                cnt++;
                max = Math.max(max,cnt);
            }
            if(ch == ')'){
                cnt--;
            }
        }
        return max;
    }
